class BusBooking:
    def __init__(self, n: int):
        self.n = n
        self.time = 0
        self.bookings = {}
        self.rows = set([])
        self.total_time = 0

    def timeToBook(self, seat):
        res = 0
        letter, row = seat[0], int(seat[1:])
        self.rows.add(int(row))
        seatType = SeatType.window if letter in ['A', 'D'] else SeatType.middle
        if seatType == SeatType.middle:
            res = 1
        elif letter == 'A':
            res = 1 + (2 if f'B{row}' in self.bookings else 0)
        elif letter == 'D':
            res = 1 + (2 if f'C{row}' in self.bookings else 0)
        return res

    def toggle(self, seat: str) -> None:
        letter, row = seat[0], int(seat[1:])
        self.rows.add(int(row))
        if seat not in self.bookings:
            t = self.timeToBook(seat)
            self.bookings[seat] = t
            self.total_time += t
            return self.bookings[seat]
        else:
            self.total_time -= self.bookings[seat]
            del self.bookings[seat]
        return 0

    def getTotalTime(self) -> int:
        return self.total_time

    def getMinTime(self) -> int:
        return len(self.bookings)
